-- Consulta analítica: top clientes com maior volume de compras
SELECT
u.id AS cliente_id,
u.nome,
u.email,
COUNT(p.id) AS total_pedidos,
SUM(p.valor_total) AS receita_gerada,
MAX(p.data_criacao) AS ultima_compra
FROM ecommerce_db.usuarios u
INNER JOIN ecommerce_db.pedidos p ON u.id = p.usuario_id
WHERE p.status = 'CONCLUIDO'
AND p.data_criacao >= '2024-01-01'
GROUP BY u.id, u.nome, u.email
HAVING receita_gerada > 1500.00
ORDER BY receita_gerada DESC
LIMIT 10;
